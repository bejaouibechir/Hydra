```python
# internal/connector/policies/auth_provider.py
class AuthProvider(ABC):  # ← Interface clé pour future migration
    @abstractmethod
    def get_headers(self) -> dict:
        """Retourne headers d'auth. Implémentation peut changer."""
        pass


# Implémentation MVP : In-Memory
class InMemoryAuthProvider(AuthProvider):
    """OAuth2 avec cache en mémoire. Temporaire mais fonctionnel."""
    
    def __init__(self, client_id: str, client_secret: str, token_url: str):
        self._token_cache = {}  # En mémoire pour MVP
    
    def get_headers(self) -> dict:
        if self._is_token_expired():
            self._refresh_token()
        return {"Authorization": f"Bearer {self._token_cache['access_token']}"}
```

**Points clés** :
- ✅ Interface `AuthProvider` = contrat stable
- ✅ Implémentation `InMemoryAuthProvider` = temporaire
- ✅ Connecteur dépend de l'interface, pas de l'implémentation
