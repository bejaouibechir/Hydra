## 🔧 Auto-discovery des plugins (Version finale)

```python
# internal/plugin_loader.py

import importlib
from pathlib import Path
from typing import Dict, Type, Any
import logging

logger = logging.getLogger(__name__)

class PluginLoader:
    """Chargeur de plugins avec auto-discovery"""

    def __init__(self, plugins_dir: Path = Path("plugins")):
        self.plugins_dir = plugins_dir
        self.loaded = {
            'connectors': {},
            'operations': {},
            'engines': {},
            'hooks': [],
            'error_handlers': {},
            'sinks': {},
            'cache': {},
            'compilers': {}
        }

    def discover_all(self):
        """Découvre tous les plugins"""
        if not self.plugins_dir.exists():
            logger.warning(f"Plugins directory not found: {self.plugins_dir}")
            return

        # Pour chaque catégorie
        for category in self.loaded.keys():
            category_path = self.plugins_dir / category
            if not category_path.exists():
                continue

            # Pour chaque fichier .py
            for plugin_file in category_path.glob("*.py"):
                if plugin_file.name.startswith("_"):
                    continue

                try:
                    self._load_plugin(category, plugin_file)
                except Exception as e:
                    logger.error(f"Failed to load plugin {plugin_file}: {e}")

    def _load_plugin(self, category: str, plugin_file: Path):
        """Charge un plugin spécifique"""
        module_name = f"plugins.{category}.{plugin_file.stem}"
        module = importlib.import_module(module_name)

        # Convention : chercher classe avec nom capitalisé
        # Ex: shopify.py → ShopifyConnector
        class_name = ''.join(word.capitalize() for word in plugin_file.stem.split('_'))

        if category == 'connectors':
            class_name += 'Connector'
        elif category == 'engines':
            class_name += 'Engine'
        elif category == 'hooks':
            class_name += 'Hook'
        # etc.

        if hasattr(module, class_name):
            plugin_class = getattr(module, class_name)

            if category == 'hooks':
                # Les hooks sont instanciés immédiatement
                self.loaded[category].append(plugin_class())
            else:
                # Les autres sont enregistrés comme classes
                plugin_id = plugin_file.stem
                self.loaded[category][plugin_id] = plugin_class

            logger.info(f"✓ Loaded {category} plugin: {plugin_id}")

    def get_connector(self, name: str) -> Type:
        """Récupère une classe de connecteur"""
        if name not in self.loaded['connectors']:
            raise ValueError(f"Connector '{name}' not found")
        return self.loaded['connectors'][name]

    def get_engine(self, name: str) -> Type:
        """Récupère une classe d'engine"""
        if name not in self.loaded['engines']:
            raise ValueError(f"Engine '{name}' not found")
        return self.loaded['engines'][name]

    def get_hooks(self) -> list:
        """Récupère tous les hooks chargés"""
        return self.loaded['hooks']

# Instance globale
plugin_loader = PluginLoader()
```

## 🔧 Auto-discovery des plugins (Version finale)

```python
# internal/plugin_loader.py

import importlib
from pathlib import Path
from typing import Dict, Type, Any
import logging

logger = logging.getLogger(__name__)

class PluginLoader:
    """Chargeur de plugins avec auto-discovery"""

    def __init__(self, plugins_dir: Path = Path("plugins")):
        self.plugins_dir = plugins_dir
        self.loaded = {
            'connectors': {},
            'operations': {},
            'engines': {},
            'hooks': [],
            'error_handlers': {},
            'sinks': {},
            'cache': {},
            'compilers': {}
        }

    def discover_all(self):
        """Découvre tous les plugins"""
        if not self.plugins_dir.exists():
            logger.warning(f"Plugins directory not found: {self.plugins_dir}")
            return

        # Pour chaque catégorie
        for category in self.loaded.keys():
            category_path = self.plugins_dir / category
            if not category_path.exists():
                continue

            # Pour chaque fichier .py
            for plugin_file in category_path.glob("*.py"):
                if plugin_file.name.startswith("_"):
                    continue

                try:
                    self._load_plugin(category, plugin_file)
                except Exception as e:
                    logger.error(f"Failed to load plugin {plugin_file}: {e}")

    def _load_plugin(self, category: str, plugin_file: Path):
        """Charge un plugin spécifique"""
        module_name = f"plugins.{category}.{plugin_file.stem}"
        module = importlib.import_module(module_name)

        # Convention : chercher classe avec nom capitalisé
        # Ex: shopify.py → ShopifyConnector
        class_name = ''.join(word.capitalize() for word in plugin_file.stem.split('_'))

        if category == 'connectors':
            class_name += 'Connector'
        elif category == 'engines':
            class_name += 'Engine'
        elif category == 'hooks':
            class_name += 'Hook'
        # etc.

        if hasattr(module, class_name):
            plugin_class = getattr(module, class_name)

            if category == 'hooks':
                # Les hooks sont instanciés immédiatement
                self.loaded[category].append(plugin_class())
            else:
                # Les autres sont enregistrés comme classes
                plugin_id = plugin_file.stem
                self.loaded[category][plugin_id] = plugin_class

            logger.info(f"✓ Loaded {category} plugin: {plugin_id}")

    def get_connector(self, name: str) -> Type:
        """Récupère une classe de connecteur"""
        if name not in self.loaded['connectors']:
            raise ValueError(f"Connector '{name}' not found")
        return self.loaded['connectors'][name]

    def get_engine(self, name: str) -> Type:
        """Récupère une classe d'engine"""
        if name not in self.loaded['engines']:
            raise ValueError(f"Engine '{name}' not found")
        return self.loaded['engines'][name]

    def get_hooks(self) -> list:
        """Récupère tous les hooks chargés"""
        return self.loaded['hooks']

# Instance globale
plugin_loader = PluginLoader()
```

# Quelle stratégie on adopte pour integrer GO?

**Ma position claire et pragmatique sur Go dans SmartETL**  
(Pas de remplacement total de Python – c’est irréaliste et destructeur – mais une intégration intelligente et progressive)

**Stratégie officielle que je propose : "Python core, Go pour les hotspots perf"**

```
Phase 1 : MVP → 100 % Python (actuel, 2025-2026)
├── Tout le core (parser, executor, DSL, connectors, DuckDB/Pandas) reste en Python
├── Avantage : vitesse de développement max, écosystème riche (pandas, duckdb, sqlalchemy)
└── Artefact compilé = option "JIT avancé" (Numba/mypyc) seulement

Phase 2 : Perf critique → Artefacts Go optionnels (2026-2027)
├── Quand un job est lent (ex: calculs lourds sur 100M+ lignes)
├── L'utilisateur active manuellement : compile: true + language: go
├── Le parser DSL → génère un programme Go autonome
│   ├── main.go templaté (à partir du pipeline.yaml)
│   ├── Utilise Arrow + Polars Go ou DuckDB Go bindings
│   └── Compile en binaire natif (artifacts/jobs/job_123_linux_amd64)
├── Exécution : compiled_artifact_runner lance le binaire Go au lieu de Python
└── Avantage : x10-50 speedup sur les hotspots, sans toucher au core

Phase 3 : Hybride mature → Core Python + Modules Go (2027+)
├── Connecteurs critiques (ex: Kafka high-throughput) → réécrits en Go comme plugins
├── Moteur de transformation → option "go_engine" qui appelle un worker Go via subprocess ou gRPC
├── Core Python reste le chef d’orchestre (parser, config, CLI, logging)
└── Jamais de migration totale : Python garde l’avantage en flexibilité/écosystème

Phase 4 : Jamais → Remplacement total par Go
├── Irréaliste : perte de pandas/DuckDB ecosystem, dev 3x plus lent
└── Contre-productif : on veut de la rapidité de dev + extensibilité
```

**Pourquoi pas un remplacement complet (contrairement à Claude qui semble plus ouvert)**

- Python = colle parfaite pour ETL (pandas, duckdb, sqlalchemy, rich ecosystem)
- Go = excellent pour perf pure et binaires, mais faible en data processing (pas de pandas équivalent mature)
- Migration totale = réécrire 80 % du projet pour 20 % de gain → non rentable
- Stratégie "hotspot only" = 80 % du gain perf avec 5 % du travail

**Verdict final (ma recommandation forte)**  
**On garde Python comme langage principal à vie.**  
**On ajoute Go uniquement comme cible de compilation optionnelle pour les jobs ultra-perf-critiques.**  
C’est le meilleur compromis vitesse dev / vitesse exécution / maintenabilité.
